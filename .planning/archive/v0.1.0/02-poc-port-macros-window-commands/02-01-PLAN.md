    ```lua
    local addonName, ns = ...

    -- Macros.lua — TLH_* macro registration with combat-lockdown deferral.
    -- Phase 2 fill: 5 named player macros using Blizzard built-in raid-marker
    -- FileDataIDs, idempotent on every login, guarded by InCombatLockdown()
    -- with a PLAYER_REGEN_ENABLED retry path (SAFE-03 / MACR-03).

    -- D-33: verbatim port of POC lines 42-78 (raid-marker FileDataIDs +
    -- idempotent Create/EditMacro pattern).
    -- D-34: printed-once "drag to action bar" hint tracked as a Lua local,
    -- NOT in the DB — printing once per session matches POC behavior.

    -- Raid marker FileDataIDs (Blizzard built-ins, no addon dependency).
    -- T has no native raid marker, so it borrows {rt1} (star) — POC line 47.
    local MACROS = {
    	{ name = "TLH_Diamond",  body = "/raid {rt3}", icon = 137003 },
    	{ name = "TLH_Triangle", body = "/raid {rt4}", icon = 137004 },
    	{ name = "TLH_Circle",   body = "/raid {rt2}", icon = 137002 },
    	{ name = "TLH_Cross",    body = "/raid {rt7}", icon = 137007 },
    	{ name = "TLH_T",        body = "/raid {rt1}", icon = 137001 },
    }

    -- Per D-34: printed-once flag is a Lua local, not in the DB.
    local macrosPrintedThisSession = false
    -- Per MACR-03: if combat blocks the first attempt, set this flag so
    -- ns:OnRegenEnabled can retry on PLAYER_REGEN_ENABLED.
    local registrationDeferred = false

    -- Re-runnable entry point. Returns true on a successful run (out of
    -- combat), false if combat blocked the attempt. Phase 3's "Recreate
    -- Macros" button (MACR-04) calls this directly.
    function ns:RegisterMacros()
    	if InCombatLockdown() then
    		registrationDeferred = true
    		return false
    	end
    	local created, updated = 0, 0
    	for _, m in ipairs(MACROS) do
    		local idx = GetMacroIndexByName(m.name)
    		if idx == 0 then
    			if CreateMacro(m.name, m.icon, m.body, false) then
    				created = created + 1
    			end
    		else
    			EditMacro(idx, m.name, m.icon, m.body)
    			updated = updated + 1
    		end
    	end
    	if not macrosPrintedThisSession and (created > 0 or updated > 0) then
    		print(string.format(
    			"|cffaa44ffTLH|r macros: %d created, %d updated. "
    				.. "Open /macro and drag them to your action bar.",
    			created,
    			updated
    		))
    		macrosPrintedThisSession = true
    	end
    	registrationDeferred = false
    	return true
    end

    -- Called by Core.lua's ADDON_LOADED dispatcher (Phase 1 already wires
    -- this). First attempt at registration; if combat blocks, the
    -- PLAYER_REGEN_ENABLED hook below retries.
    function ns:InitMacros()
    	if not ns:RegisterMacros() then
    		-- Combat blocked the initial attempt; arm the retry hook.
    		local f = CreateFrame("Frame")
    		f:RegisterEvent("PLAYER_REGEN_ENABLED")
    		f:SetScript("OnEvent", function(self, event)
    			if event == "PLAYER_REGEN_ENABLED" and registrationDeferred then
    				ns:RegisterMacros()
    				if not registrationDeferred then
    					self:UnregisterEvent("PLAYER_REGEN_ENABLED")
    				end
    			end
    		end)
    	end
    end

    -- Public hook that other modules can call if they ever need to drive
    -- the regen retry from outside (currently unused; provided for
    -- module-API surface symmetry with ns:RegisterMacros).
    function ns:OnRegenEnabled()
    	if registrationDeferred then
    		ns:RegisterMacros()
    	end
    end
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Player UI → Macro body | Macro bodies (`/raid {rt#}`) are author-controlled string literals — no user-supplied input ever flows into `CreateMacro`. |
| Combat lockdown → Macro write | Blizzard's lockdown blocks `CreateMacro`/`EditMacro` mid-combat; addon must self-defer. |
| SavedVariables → Lua state | `db.window.alpha` is read by Phase 2 plan 02-02 from disk; tampering would only change window opacity (range-clamped by Phase 3 slider). Negligible attack surface. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | T (Tampering) | `db.window.alpha` field | accept | User-editable SavedVariables file is the WoW-standard config surface. Phase 2 reads the value once at frame creation; out-of-range values produce a dim or fully-transparent window — not an exploit, just a self-inflicted UX issue. Phase 3's slider will clamp to [0.20, 1.00]. |
| T-02-02 | E (Elevation of Privilege) | `CreateMacro`/`EditMacro` during combat | mitigate | `InCombatLockdown()` guard at the top of `ns:RegisterMacros` short-circuits the call. `PLAYER_REGEN_ENABLED` retry hook re-attempts only when combat is over. The Blizzard secure environment enforces this at the API level even if our guard fails — we add the explicit guard for clean UX (no Lua error pop-up). |
| T-02-03 | I (Information Disclosure) | Printed-once hint string | accept | The hint is a hard-coded literal containing only the addon name and macro instruction. No user data, no PII, no DB contents are concatenated into the print. Output goes to the local chat frame only. |
| T-02-04 | T (Tampering) | `MACROS` table at module top | accept | `MACROS` is a Lua local; not exposed on `ns` and not reachable by other addons through the global table. Modifying it would require editing `Macros.lua` on disk, which is equivalent to modifying any other addon code — out-of-scope per WoW addon threat model. |
</threat_model>
