    if not inInstance or instanceType ~= "raid" then
        ns:HideWindow()
        return

    local instanceID = select(8, GetInstanceInfo())
    if DEBUG_ZONE_INFO then

                tostring(mapID),

    if instanceID == LURA_RAID_INSTANCE_ID then
        ns:ShowWindow()
    else
        ns:HideWindow()
    end
