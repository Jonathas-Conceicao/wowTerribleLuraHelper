<chat_filter_rule>
<!-- The exact channel-filter logic per D-05 + D-31. -->

`db.listenChannels` keys (set in Phase 1's schema):
  SAY, RAID, RAID_LEADER, RAID_WARNING, INSTANCE_CHAT, INSTANCE_CHAT_LEADER

Event names (CHAT_MSG_ prefix is 9 chars; `event:sub(10)` strips it):
  CHAT_MSG_SAY                 → "SAY"
  CHAT_MSG_RAID                → "RAID"
  CHAT_MSG_RAID_LEADER         → "RAID_LEADER"
  CHAT_MSG_RAID_WARNING        → "RAID_WARNING"
  CHAT_MSG_INSTANCE_CHAT       → "INSTANCE_CHAT"
  CHAT_MSG_INSTANCE_CHAT_LEADER → "INSTANCE_CHAT_LEADER"

Filter check at the TOP of the chat handler (before any `msg` work):
```lua
if not ns.db.listenChannels[event:sub(10)] then
    return  -- channel disabled; drop event silently
end
```

NOTE: `event:sub(10)` is a string operation on `event`, NOT on `msg`. SAFE-02 only forbids string ops on `msg` (the chat message body), not on `event` (the event name). This is explicitly safe.
</chat_filter_rule>

    -- Window.lua — smile-arc helper window: 5 positional rune slots, BOSS/TANK labels,
    -- lock/unlock, drag-position persistence, 20s self-clear, in-memory sequence.
    -- Slot text MUST be set directly from
    -- C_ChatInfo.ReplaceIconAndGroupExpressions(msg, nil, false) — never index msg in Lua.

    -- D-23..D-25 visibility model, D-27 in-memory sequence, D-28 20s timeout,
    -- D-30..D-32 channel filter + event gating).

    -- ============================================================
    -- Constants (D-28, D-29: hardcoded for v1, named for future v2 promotion)
    -- ============================================================
    local INACTIVITY_TIMEOUT = 20

    local W, H = 380, 320
    local SLOT_SIZE = 64
    local ICON_SIZE = SLOT_SIZE - 8

    -- Smile-arc geometry verbatim from POC (D-18). Offsets are CENTER-relative
    -- to bossLabel.
    local SLOT_POS = {
    	[1] = {  110,  32 },
    	[2] = {   80, -28 },
    	[3] = {    0, -60 },
    	[4] = {  -80, -28 },
    	[5] = { -110,  32 },
    }

    local CHAT_EVENTS = {
    	"CHAT_MSG_SAY",
    	"CHAT_MSG_RAID",
    	"CHAT_MSG_RAID_LEADER",
    	"CHAT_MSG_RAID_WARNING",
    	"CHAT_MSG_INSTANCE_CHAT",
    	"CHAT_MSG_INSTANCE_CHAT_LEADER",
    }

    -- ============================================================
    -- State (in-memory only per D-27 — NOT in TerribleLuraHelperDB)
    -- ============================================================
    local sequence = {}
    local clearTimer
    local positionApplied = false  -- saved position is applied on first show only

    -- Forward declarations so handlers below can call helpers above.
    local win, slotFrames, lockBtn
    local FillSlot, ClearAll, ScheduleClear, ManualClear
    local applyLockState, applySavedPosition, persistPosition

    -- ============================================================
    -- Frame construction (called from ns:InitWindow)
    -- ============================================================
    local function CreateWindow()
    	-- D-17: BasicFrameTemplateWithInset replaces the POC's BackdropTemplate.
    	-- The template provides title bar, close button, rock+marble backdrop.
    	win = CreateFrame("Frame", "TerribleLuraHelperWindow", UIParent, "BasicFrameTemplateWithInset")
    	win:SetSize(W, H)
    	win:EnableMouse(true)
    	win:SetClampedToScreen(true)
    	win:Hide()  -- WIN-04: hidden by default on every login/reload

    	-- Initial anchor — overridden by applySavedPosition on first show.
    	win:SetPoint("CENTER", UIParent, "CENTER", 200, 80)

    	-- Apply scale and alpha at creation (D-36: alpha read once; live update is Phase 3).
    	win:SetScale(ns.db.window.scale or 1.00)
    	win:SetAlpha(ns.db.window.alpha or 1.00)

    	-- D-19: keep template's default title color. Set text only.
    	if win.TitleText then
    		win.TitleText:SetText("Terrible L'ura Helper")
    	end

    	-- Template's CloseButton — override OnClick to hide (not destroy) the frame.
    	if win.CloseButton then
    		win.CloseButton:SetScript("OnClick", function() win:Hide() end)
    	end

    	-- D-14..D-16: lock button immediately LEFT of CloseButton, 20x20.
    	lockBtn = CreateFrame("Button", nil, win)
    	lockBtn:SetSize(20, 20)
    	lockBtn:SetPoint("TOPRIGHT", win.CloseButton, "TOPLEFT", 2, 0)
    	lockBtn:SetHighlightTexture("Interface\\Buttons\\UI-Common-MouseHilight", "ADD")
    	lockBtn:SetScript("OnClick", function() ns:ToggleLocked() end)
    	-- Texture is set by applyLockState below.

    	-- ============================================================
    	-- Boss view + smile-arc slots (port verbatim from POC lines 121-178; D-18)
    	-- ============================================================
    	local bossView = CreateFrame("Frame", nil, win)
    	bossView:SetSize(W - 40, 220)
    	bossView:SetPoint("TOP", win, "TOP", 0, -38)

    	local bossLabel = bossView:CreateFontString(nil, "OVERLAY")
    	bossLabel:SetFont("Fonts\\FRIZQT__.TTF", 18, "OUTLINE")
    	bossLabel:SetPoint("CENTER", bossView, "CENTER", 0, 8)
    	bossLabel:SetText("BOSS")

    	local tankLabel = bossView:CreateFontString(nil, "OVERLAY")
    	tankLabel:SetFont("Fonts\\FRIZQT__.TTF", 14, "OUTLINE")
    	tankLabel:SetPoint("CENTER", bossLabel, "CENTER", 0, 60)
    	tankLabel:SetText("TANK")

    	slotFrames = {}
    	for i = 1, 5 do
    		local slot = CreateFrame("Frame", nil, bossView, "BackdropTemplate")
    		slot:SetSize(SLOT_SIZE, SLOT_SIZE)
    		slot:SetPoint("CENTER", bossLabel, "CENTER", SLOT_POS[i][1], SLOT_POS[i][2])

    			edgeFile = "Interface\\Buttons\\WHITE8x8",
    			edgeSize = 1,
    		})
    		slot:SetBackdropColor(0.05, 0.05, 0.1, 0.7)
    		slot:SetBackdropBorderColor(0.3, 0.2, 0.5, 0.7)

    		local fs = slot:CreateFontString(nil, "OVERLAY")
    		fs:SetFont("Fonts\\FRIZQT__.TTF", ICON_SIZE, "OUTLINE")
    		fs:SetPoint("CENTER")
    		fs:SetText("")

    		idx:SetText(tostring(i))
    		idx:SetTextColor(0.65, 0.65, 0.7, 0.85)

    		slotFrames[i] = slot
    	end

    	-- ============================================================
    	-- Drag handlers — bound regardless of locked state; SetMovable
    	-- is the actual gate (D-15, UI-SPEC §3.2).
    	-- ============================================================
    	win:SetScript("OnDragStart", function(self)
    		if self:IsMovable() then self:StartMoving() end
    	end)
    	win:SetScript("OnDragStop", function(self)
    		self:StopMovingOrSizing()
    		persistPosition()
    	end)

    	-- Apply current locked/unlocked state to drag bindings + button texture.
    	applyLockState()
    end

    -- ============================================================
    -- Slot helpers (port verbatim from POC FillSlot/ClearAll)
    -- ============================================================
    FillSlot = function(i, msg)
    	local slot = slotFrames[i]
    	if not slot then return end
    	-- Pass msg opaquely. SetText is C-level and does not index msg.
    	slot.fs:SetText(msg)
    	slot:SetBackdropBorderColor(0.85, 0.4, 1.0, 0.9)
    end

    ClearAll = function()
    	wipe(sequence)
    	for i = 1, 5 do
    		slotFrames[i].fs:SetText("")
    		slotFrames[i]:SetBackdropBorderColor(0.3, 0.2, 0.5, 0.7)
    	end
    end

    -- ============================================================
    -- Lock state (D-14..D-16, UI-SPEC §3.1)
    -- ============================================================
    applyLockState = function()
    	local locked = ns.db.window.locked
    	if locked then
    		win:SetMovable(false)
    		win:RegisterForDrag()
    		lockBtn:SetNormalTexture("Interface\\Buttons\\LockButton-Locked-Up")
    		lockBtn:SetPushedTexture("Interface\\Buttons\\LockButton-Locked-Down")
    	else
    		win:SetMovable(true)
    		win:RegisterForDrag("LeftButton")
    		lockBtn:SetNormalTexture("Interface\\Buttons\\LockButton-Unlocked-Up")
    		lockBtn:SetPushedTexture("Interface\\Buttons\\LockButton-Unlocked-Down")
    	end
    end

    function ns:ToggleLocked()
    	ns.db.window.locked = not ns.db.window.locked
    	applyLockState()
    end

    -- ============================================================
    -- Position persistence (D-20..D-22, WIN-09)
    -- ============================================================
    persistPosition = function()
    	local point, relativeTo, relativePoint, x, y = win:GetPoint()
    	ns.db.window.position = {
    		point,
    		(relativeTo and relativeTo.GetName and relativeTo:GetName()) or "UIParent",
    		relativePoint,
    		x,
    		y,
    	}
    end

    applySavedPosition = function()
    	if positionApplied then return end
    	positionApplied = true
    	local pos = ns.db.window.position
    	win:ClearAllPoints()
    	if pos then
    		win:SetPoint(pos[1], _G[pos[2]] or UIParent, pos[3], pos[4], pos[5])
    	else
    		win:SetPoint("CENTER", UIParent, "CENTER", 200, 80)
    	end
    end

    -- ============================================================
    -- ns:InitWindow — Phase 1 dispatcher entry point
    -- ============================================================
    function ns:InitWindow()
    	CreateWindow()
    	-- Window stays hidden until /lura show. WIN-04 / D-25.
    end
